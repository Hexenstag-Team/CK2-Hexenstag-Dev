reiklander_nulnian_war_cb = {
    name = CB_NAME_REIKLANDER_NULNIAN_WAR_CB
    war_name = WAR_NAME_REIKLANDER_NULNIAN_WAR_CB
    sprite = 1
    truce_days = 3650
    hostile_against_others = yes
    is_permanent = no
	battle_warscore_mult = 2
	max_attacker_battle_score = 100
	allow_whitepeace = no
	attacker_unoccupied_warscore  = yes

    can_use = {
		ROOT = {
			has_landed_title = e_reikland
		}
		FROM = {
			has_landed_title = e_nuln
		}
    }

	can_use_title = {
		ROOT = {
			primary_title = {
				has_landed_title = e_reikland
			}
		}
		FROM = {
			primary_title = {
				has_landed_title = e_nuln
			}
		}
    }

	is_valid = {
		# The defender is not in any black crusades, otherwise the joining effect for defenders will join the wrong war
		FROM = {
			NOT = {
				any_war = {
					using_cb = new_crusade
				}
			}
		}
		ROOT = {
			primary_title = { higher_tier_than = KING }
		}
		FROM = {
			primary_title = { higher_tier_than = KING }
		}
	}

	is_valid_title = {
		FROM = {
			has_landed_title = e_nuln
		}
		ROOT = {
			has_landed_title = e_reikland
		}
	}

    on_success = {
        ROOT = {
            prestige = 1000
        }
        any_attacker = {
            limit = {
				character = ROOT
			}
            participation_scaled_prestige = 1000
        }
        any_attacker = {
            limit = {
				NOT = {
					character = ROOT
				}
			}
            hidden_tooltip = {
				participation_scaled_prestige = 100
			}
        }
        FROM = {
            prestige = -1000
			any_vassal = {
				if = {
					limit = {
						primary_title = {
							higher_tier_than = DUKE
						}
					}
					set_defacto_liege = ROOT
					hidden_tooltip = {
						opinion = {
							modifier = opinion_traitor
							who = ROOT
							months = 160
						}
					}
				}
			}
			primary_title = {
				if = {
					limit = {
						higher_tier_than = KING
					}
					destroy_landed_title = THIS
				}
			}
			set_defacto_liege = ROOT
			hidden_tooltip = {
				opinion = {
					modifier = opinion_regicide_traitor
					who = ROOT
					months = 1200
				}
			}
		}
		set_global_flag = reiklander_nulnian_war_reikland
		k_averland = {
			clr_imperial_flags = yes
			set_title_flag = vassal_of_reikland
		}
		k_stirland = {
			clr_imperial_flags = yes
			set_title_flag = vassal_of_reikland
		}
		k_wissenland = {
			clr_imperial_flags = yes
			set_title_flag = vassal_of_reikland
		}
		d_nuln = {
			clr_imperial_flags = yes
		}
    }

    on_fail = { }

    on_reverse_demand = {
		FROM = {
            prestige = 1000
        }
        any_defender = {
            limit = {
				character = FROM
			}
            participation_scaled_prestige = 1000
        }
        any_defender = {
            limit = {
				NOT = {
					character = FROM
				}
			}
            hidden_tooltip = {
				participation_scaled_prestige = 100
			}
        }
        ROOT = {
            prestige = -1000
			any_vassal = {
				if = {
					limit = {
						primary_title = {
							higher_tier_than = DUKE
						}
					}
					set_defacto_liege = FROM
					hidden_tooltip = {
						opinion = {
							modifier = opinion_traitor
							who = FROM
							months = 160
						}
					}
				}
			}
			primary_title = {
				if = {
					limit = {
						higher_tier_than = KING
					}
					destroy_landed_title = THIS
				}
			}
			set_defacto_liege = FROM
			hidden_tooltip = {
				opinion = {
					modifier = opinion_regicide_traitor
					who = FROM
					months = 1200
				}
			}
		}
		set_global_flag = reiklander_nulnian_war_nuln
		k_reikland = {
			clr_imperial_flags = yes
			set_title_flag = vassal_of_nuln
		}
		k_carroburg_middenland = {
			clr_imperial_flags = yes
			set_title_flag = vassal_of_nuln
		}
		k_mootland = {
			clr_imperial_flags = yes
			set_title_flag = vassal_of_nuln
		}
    }

    attacker_ai_victory_worth = {
        factor = -1 # always accept
    }

    attacker_ai_defeat_worth = {
        factor = 100
    }

    defender_ai_victory_worth = {
        factor = -1 # always accept
    }

    defender_ai_defeat_worth = {
        factor = 100
    }
}

sylvania_boogaloo = {
	name = CB_NAME_SYLVANIA_BOOGALOO
	war_name = WAR_NAME_SYLVANIA_BOOGALOO
	sprite = 27
	truce_days = 1825
	hostile_against_others = yes
	is_permanent = yes
	check_all_titles = yes # if permanent, setting this to true will check against all of someones titles, including vassal held titles
	can_ask_to_join_war = no
	allowed_to_target_tributaries = no
	infamy_modifier = 5

	sort_priority = 900

	can_use_gui = {
		ROOT = {
			piety = 250
		}
	}

	on_add = {
	}

	can_use = {
		ROOT = {
			capital_scope = { region = world_old_empire_sylvania }
		}
	}

	can_use_title = {
		capital_scope = { region = world_old_empire_sylvania }
	}

	is_valid = {
		# The defender is not in any black crusades, otherwise the joining effect for defenders will join the wrong war
		FROM = {
			NOT = {
				any_war = {
					using_cb = new_crusade
				}
			}
		}
		ROOT = {
			NOT = { pacifist = yes }
			NOT = { same_realm = FROM }
		}
	}

	on_success_title = {
		usurp_title_plus_barony_if_unlanded = { target = ROOT type = invasion }
	}

	on_fail_title = {
		ROOT = {
			show_scope_change = no
			prestige = -100
		}
		FROM = {
			show_scope_change = no
			participation_scaled_prestige = 50
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 50 }
		}
	}

	on_reverse_demand = {
		ROOT = {
			show_scope_change = no
			prestige = -200
			transfer_scaled_wealth = {
				to = FROM
				value = 4.0
			}
		}
		FROM = {
			show_scope_change = no
			participation_scaled_prestige = 100
		}
		any_defender = {
			limit = { NOT = { character = FROM } }
			hidden_tooltip = { participation_scaled_prestige = 100 }
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 1

		modifier = {
			factor = 0
			ROOT = {
				OR = {
					trait = just
					trait = content
				}
			}
		}
		modifier = {
			factor = 0
			ROOT = {
				NOR = {
					trait = arbitrary
					trait = ambitious
				}
			}
		}
	}
}
