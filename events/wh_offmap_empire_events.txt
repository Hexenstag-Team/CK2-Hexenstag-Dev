character_event = {	# Stability is initialized at 300 and recovers by 1 each year.
	id = wh_offmap_empire_events.100
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		FROM = { is_offmap_tag = offmap_the_empire }
	}

	immediate = {
		if = {
			limit = {
				FROM = { NOT = { has_offmap_flag = the_empire_stability_initialized } }
			}
			set_variable = { which = the_empire_stability value = 300 }
			FROM = {
				set_offmap_flag = the_empire_stability_initialized
				set_status = status_golden_age
			}
		}
		else = {
			if = {
				limit = { NOT = { check_variable = { which = the_empire_stability value >= 300 } } }
				change_variable = { which = the_empire_stability value = 1 }
			}
			if = {
				limit = { check_variable = { which = the_empire_stability value >= 300 } }
				set_variable = { which = the_empire_stability value = 300 }
			}
			update_the_empire_stability_status_effect = yes
		}
	}
}
